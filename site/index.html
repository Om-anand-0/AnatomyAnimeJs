Media queries
animate('.square', { rotate: 90, loop: true, ease: 'inOutExpo', });
animate('.shape', { x: random(-100, 100), y: random(-100, 100), rotate: random(-180, 180), duration: random(500, 1000), composition: 'blend', });
animate('.ring', { rotate: 360, delay: stagger(100), autoplay: onScroll({ sync: true, }), });
const options = { grid: [13, 13], from: 'center', }; createTimeline() .add('.dot', { scale: stagger([1.1, .75], options), }, stagger(200, options));
animate(svg.createDrawable('.circuit'), { draw: ['0 0', '0 1', '1 1'], ease: 'inOutQuad', loop: true, });
createDraggable('.circle', { releaseEase: createSpring({ stiffness: 120, damping: 6, }), });
createTimeline() .add('.tick', { y: '-=6', duration: 50, }, stagger(10)) .add('.ticker', { rotate: 360, duration: 1920, }, '<');
createScope({ mediaQueries: { portrait: '(orientation: portrait)', } }).add(({ matches }) => { animate('.circle', { rotate: matches.portrait ? 90 : 0, }); });
